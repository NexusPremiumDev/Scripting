function NexusESPLibrary:AddESP(Cfg)
    local Obj = Cfg.Object
    if not Obj then return end
    if self.Cache[Obj] then return end

    local Root = NexusESPGetRoot(Obj)
    if not Root then
        task.spawn(function()
            local Attempts = 0
            while not Root and Attempts < 60 do
                Root = NexusESPGetRoot(Obj)
                Attempts = Attempts + 1
                task.wait(0.5)
            end
            if Root and Obj.Parent then
                self:AddESP(Cfg)
            end
        end)
        return
    end

    local Billboard = Instance.new("BillboardGui")
    Billboard.Name = HttpService:GenerateGUID(false)
    Billboard.Size = UDim2.new(0, 250, 0, 60)
    Billboard.AlwaysOnTop = true
    Billboard.Adornee = Root
    Billboard.MaxDistance = self.Settings.MaxDistance
    Billboard.Parent = Obj

    local Label = Instance.new("TextLabel")
    Label.BackgroundTransparency = 1
    Label.Size = UDim2.new(1, 0, 1, 0)
    Label.TextColor3 = Cfg.Color
    Label.Font = self.Settings.Font
    Label.TextSize = self.Settings.TextSize
    Label.TextStrokeTransparency = 0
    Label.TextTransparency = 1
    Label.Text = Cfg.Text
    Label.Parent = Billboard

    local Highlight = Instance.new("Highlight")
    Highlight.Adornee = Obj
    Highlight.FillColor = Cfg.Color
    Highlight.FillTransparency = 1
    Highlight.OutlineTransparency = 1
    Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    Highlight.Parent = Obj

    local Line = Instance.new("Frame")
    Line.AnchorPoint = Vector2.new(0.5, 0.5)
    Line.BorderSizePixel = 0
    Line.BackgroundColor3 = Cfg.Color
    Line.Visible = true
    Line.BackgroundTransparency = 1
    Line.Parent = NexusESPScreen

    local Arrow = Instance.new("ImageLabel")
    Arrow.BackgroundTransparency = 1
    Arrow.AnchorPoint = Vector2.new(0.5, 0.5)
    Arrow.Size = UDim2.new(0, 32, 0, 32)
    Arrow.Image = "rbxassetid://10526559647"
    Arrow.ImageColor3 = Cfg.Color
    Arrow.Visible = false
    Arrow.Parent = NexusESPScreen

    local FT = self.Settings.FadeTime
    if FT > 0 then
        TweenService:Create(Label, self.TweenInfoFadeIn, {TextTransparency = 0}):Play()
        TweenService:Create(Highlight, self.TweenInfoFadeIn, {FillTransparency = 0.6}):Play()
        TweenService:Create(Line, self.TweenInfoFadeIn, {BackgroundTransparency = 0}):Play()
    else
        Label.TextTransparency = 0
        Highlight.FillTransparency = 0.6
        Line.BackgroundTransparency = 0
    end

    self.Cache[Obj] = {
        Tag = Billboard,
        Text = Label,
        Line = Line,
        Box = Highlight,
        Arrow = Arrow,
        Name = Cfg.Text,
        Color = Cfg.Color,
        Root = Root,
    }
end

function NexusESPLibrary:RemoveESP(Obj)
    local Data = self.Cache[Obj]
    if not Data then return end
    self.Cache[Obj] = nil

    local FT = self.Settings.FadeTime
    if FT > 0 then
        local DelayId = HttpService:GenerateGUID(false)
        self.ActiveDelays[DelayId] = true

        TweenService:Create(Data.Text, self.TweenInfoFadeOut, {TextTransparency = 1}):Play()
        TweenService:Create(Data.Box, self.TweenInfoFadeOut, {FillTransparency = 1}):Play()
        TweenService:Create(Data.Line, self.TweenInfoFadeOut, {BackgroundTransparency = 1}):Play()

        task.delay(FT, function()
            if self.ActiveDelays[DelayId] then
                self.ActiveDelays[DelayId] = nil
                if Data.Tag then Data.Tag:Destroy() end
                if Data.Line then Data.Line:Destroy() end
                if Data.Box then Data.Box:Destroy() end
                if Data.Arrow then Data.Arrow:Destroy() end
            end
        end)
    else
        if Data.Tag then Data.Tag:Destroy() end
        if Data.Line then Data.Line:Destroy() end
        if Data.Box then Data.Box:Destroy() end
        if Data.Arrow then Data.Arrow:Destroy() end
    end
end

function NexusESPLibrary:UpdateESP(Obj, NewText, NewColor)
    local Data = self.Cache[Obj]
    if Data then
        if NewText and Data.Name ~= NewText then
            Data.Name = NewText
            Data.Text.Text = NewText
        end
        if NewColor and Data.Color ~= NewColor then
            Data.Color = NewColor
            Data.Text.TextColor3 = NewColor
            Data.Box.FillColor = NewColor
            Data.Line.BackgroundColor3 = NewColor
            Data.Arrow.ImageColor3 = NewColor
        end
    else
        self:AddESP({Object = Obj, Text = NewText or "ESP", Color = NewColor or Color3.new(1, 1, 1)})
    end
end

function NexusESPLibrary:BulkUpdateESP(UpdateTable)
    for Obj, Info in pairs(UpdateTable) do
        local Data = self.Cache[Obj]
        if Data then
            local NeedsColorUpdate = false
            if Info.Text and Data.Name ~= Info.Text then
                Data.Name = Info.Text
                Data.Text.Text = Info.Text
            end
            if Info.Color and Data.Color ~= Info.Color then
                Data.Color = Info.Color
                NeedsColorUpdate = true
            end
            if NeedsColorUpdate then
                local Color = Info.Color
                Data.Text.TextColor3 = Color
                Data.Box.FillColor = Color
                Data.Line.BackgroundColor3 = Color
                Data.Arrow.ImageColor3 = Color
            end
        else
            if Info.Text and Info.Color then
                self:AddESP({Object = Obj, Text = Info.Text, Color = Info.Color})
            end
        end
    end
end

function NexusESPLibrary:ClearAllESP()
    local Objects = {}
    for Obj in pairs(self.Cache) do
        table.insert(Objects, Obj)
    end
    for i = 1, #Objects do
        self:RemoveESP(Objects[i])
    end
end

function NexusESPLibrary:Unload()
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end

    for DelayId in pairs(self.ActiveDelays) do
        self.ActiveDelays[DelayId] = nil
    end

    for Obj in pairs(self.Cache) do
        local Data = self.Cache[Obj]
        self.Cache[Obj] = nil
        if Data then
            if Data.Tag then Data.Tag:Destroy() end
            if Data.Line then Data.Line:Destroy() end
            if Data.Box then Data.Box:Destroy() end
            if Data.Arrow then Data.Arrow:Destroy() end
        end
    end

    if NexusESPScreen then
        NexusESPScreen:Destroy()
    end
end

local NexusESPCamera = workspace.CurrentCamera
local NexusESPScreenCenter = NexusESPCamera.ViewportSize / 2
local NexusESPFOVScale = NexusESPCamera.FieldOfView / 70
local RainbowColor = Color3.fromHSV(0, 1, 1)
local LastUpdateTick = tick()

NexusESPLibrary.Connection = RunService.Heartbeat:Connect(function()
    local RenderInterval = 1 / NexusESPLibrary.Settings.RenderTime
    if tick() - LastUpdateTick < RenderInterval then return end
    LastUpdateTick = tick()

    NexusESPCamera = workspace.CurrentCamera
    local ViewportSize = NexusESPCamera.ViewportSize
    NexusESPScreenCenter = ViewportSize / 2
    NexusESPFOVScale = NexusESPCamera.FieldOfView / 70
    RainbowColor = Color3.fromHSV(tick() % 5 / 5, 1, 1)
    local CameraCFrame = NexusESPCamera.CFrame

    for Object, Data in pairs(NexusESPLibrary.Cache) do
        if not Object or not Object.Parent then
            NexusESPLibrary:RemoveESP(Object)
            continue
        end

        local RootPart = Data.Root
        if not RootPart or not RootPart.Parent then
            Data.Root = NexusESPGetRoot(Object)
            RootPart = Data.Root
            if not RootPart then continue end
        end

        local ScreenPosition, IsOnScreen = NexusESPCamera:WorldToViewportPoint(RootPart.Position)
        local DisplayColor = NexusESPLibrary.Settings.Rainbow and RainbowColor or Data.Color
        local CalculatedDistance = Data.Tag.CurrentDistance

        if CalculatedDistance > NexusESPLibrary.Settings.MaxDistance then
            Data.Tag.Enabled = false
            Data.Text.Visible = false
            Data.Box.Enabled = false
            Data.Line.Visible = false
            Data.Arrow.Visible = false
            continue
        end

        if IsOnScreen then
            local ESPMode = NexusESPLibrary.Settings.ESPMode
            local ShowHighlight = ESPMode == "Highlight" or ESPMode == "Highlight/Text"
            local ShowText = ESPMode == "Text" or ESPMode == "Highlight/Text"

            Data.Tag.Enabled = true
            Data.Arrow.Visible = false

            if ShowText then
                Data.Text.Visible = true
                Data.Text.TextColor3 = DisplayColor
                Data.Text.TextSize = NexusESPLibrary.Settings.TextSize * NexusESPFOVScale
                local DistanceStr = ""
                if NexusESPLibrary.Settings.Distance then
                    DistanceStr = "\n[" .. math.floor(CalculatedDistance) .. "m]"
                end
                Data.Text.Text = Data.Name .. DistanceStr
            else
                Data.Text.Visible = false
            end

            if ShowHighlight then
                Data.Box.Enabled = true
                Data.Box.FillColor = DisplayColor
            else
                Data.Box.Enabled = false
            end

            if NexusESPLibrary.Settings.Tracers and CalculatedDistance < 1500 then
                local ScreenPoint = Vector2.new(ScreenPosition.X, ScreenPosition.Y)
                local OriginPoint = Vector2.new(NexusESPScreenCenter.X, ViewportSize.Y)
                local VectorDifference = ScreenPoint - OriginPoint

                Data.Line.Visible = true
                Data.Line.BackgroundColor3 = DisplayColor
                Data.Line.Size = UDim2.new(0, VectorDifference.Magnitude, 0, 1)
                Data.Line.Position = UDim2.new(0, OriginPoint.X + VectorDifference.X / 2, 0, OriginPoint.Y + VectorDifference.Y / 2)
                Data.Line.Rotation = math.deg(math.atan2(VectorDifference.Y, VectorDifference.X))
            else
                Data.Line.Visible = false
            end
        else
            Data.Tag.Enabled = false
            Data.Text.Visible = false
            Data.Box.Enabled = false
            Data.Line.Visible = false

            if NexusESPLibrary.Settings.Arrows then
                local ObjectRelativeSpace = CameraCFrame:PointToObjectSpace(RootPart.Position)
                local DirectionAngle = math.atan2(-ObjectRelativeSpace.X, -ObjectRelativeSpace.Z)
                local Radius = ViewportSize.Y * 0.4

                Data.Arrow.Visible = true
                Data.Arrow.ImageColor3 = DisplayColor
                Data.Arrow.Position = UDim2.new(0, NexusESPScreenCenter.X + math.sin(DirectionAngle) * Radius, 0, NexusESPScreenCenter.Y + math.cos(DirectionAngle) * Radius)
                Data.Arrow.Rotation = math.deg(DirectionAngle)
            else
                Data.Arrow.Visible = false
            end
        end
    end
end)

local function NexusESPGetRoot(obj)
    if not obj then return nil end
    if obj:IsA("BasePart") then return obj end
    if obj:IsA("Model") then
        return obj.PrimaryPart
            or obj:FindFirstChild("HumanoidRootPart")
            or obj:FindFirstChildWhichIsA("BasePart", true)
    end
    return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function matches(kind, obj)
    local map = NAMES[kind]
    if not map then return false end
    if map[obj.Name] then return true end

    -- DOORS rooms often contain nested objects with useful names.
    for _, d in ipairs(obj:GetDescendants()) do
        if map[d.Name] then return true end
    end
    return false
end

local function objectText(kind, obj)
    local labels = {
        Door="Door", Objective="Objective", HidingPlace="Hiding Place",
        GateLever="Gate Lever", Books="Book", Breakers="Breaker",
        Gold="Gold", Entity="Entity", Ladder="Ladder", Fuse="Fuse",
    }
    local base = labels[kind] or kind
    if kind == "Entity" then
        local entityNames = {
            RushMoving="Rush", AmbushMoving="Ambush", A60="A-60", A120="A-120",
            Eyes="Eyes", Lookman="Eyes", BackdoorRush="Blitz",
            BackdoorLookman="Lookman", JeffTheKiller="Jeff",
            FigureRig="Figure", SeekMoving="Seek",
        }
        return entityNames[obj.Name] or base
    end
    return base
end

local function add(kind, obj)
    if not Enabled[kind] or not obj or not obj.Parent then return end
    if Tracked[obj] then return end
    local root = NexusESPGetRoot(obj)
    if not root then return end

    Tracked[obj] = {kind=kind}
    NexusESPLibrary:AddESP({
        Object = obj,
        Text = objectText(kind, obj),
        Color = ESP_COLORS[kind],
        NexusKind = kind,
    })
end

local function remove(obj)
    if not obj then return end
    Tracked[obj] = nil
    pcall(function() NexusESPLibrary:RemoveESP(obj) end)
end

local function scan(kind)
    if not Enabled[kind] then return end

    local seen = {}
    local rooms = workspace:FindFirstChild("CurrentRooms")

    if rooms then
        local current = tonumber(NexusESPLP:GetAttribute("CurrentRoom")) or 0
        for _, room in ipairs(rooms:GetChildren()) do
            local n = tonumber(room.Name)
            if n == nil or math.abs(n - current) <= 2 then
                for _, obj in ipairs(room:GetDescendants()) do
                    if matches(kind, obj) and NexusESPGetRoot(obj) then
                        -- Prefer the meaningful parent when nested parts match.
                        local target = obj
                        if not (obj:IsA("Model") or obj:IsA("BasePart")) then
                            target = obj:FindFirstAncestorOfClass("Model") or obj
                        end
                        if NexusESPGetRoot(target) then
                            seen[target] = true
                            add(kind, target)
                        end
                    end
                end
            end
        end
    end

    -- Entities can live outside CurrentRooms.
    if kind == "Entity" then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if matches(kind, obj) and NexusESPGetRoot(obj) then
                seen[obj] = true
                add(kind, obj)
            end
        end
    end

    -- Some objectives/loot also appear outside a room container.
    if kind == "Gold" or kind == "Objective" or kind == "Fuse" then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if matches(kind, obj) and NexusESPGetRoot(obj) then
                seen[obj] = true
                add(kind, obj)
            end
        end
    end

    for obj, data in pairs(Tracked) do
        if data.kind == kind and (not seen[obj] or not obj.Parent) then
            remove(obj)
        end
    end
end

local function clearKind(kind)
    for obj, data in pairs(Tracked) do
        if data.kind == kind then
            remove(obj)
        end
    end
end

function NexusSetESP(kind, value)
    Enabled[kind] = value == true
    if Enabled[kind] then
        -- Immediate first scan; no delayed GUI callback dependency.
        scan(kind)
        task.defer(function() scan(kind) end)
    else
        clearKind(kind)
    end
end

function PathTo(Pos)
local Character = LocalPlayer.Character
local Root = Character and Character:FindFirstChild("HumanoidRootPart")
local Humanoid = Character and Character:FindFirstChild("Humanoid")
if not Root or not Humanoid then return end
local p = PathfindingService:CreatePath({
AgentRadius = 2,
AgentHeight = 5,
AgentCanJump = true,
WaypointSpacing = 2,
AgentCanSprint = false,
AgentMaxSlope = 45,
AgentJumpHeight = 0
})
local AdjustedPos = Pos and Pos +Vector3.new(0, 0, 1)
p:ComputeAsync(Root.Position, AdjustedPos)
if  p.Status ~= Enum.PathStatus.Success then return end
for _, waypoint in ipairs(p:GetWaypoints()) do
if not Toggles.AutoRooms.Value or Library.Unloaded then break end
local Dist = (Root.Position - waypoint.Position).Magnitude
if Dist >= 2 then
Humanoid:MoveTo(waypoint.Position)
Humanoid.MoveToFinished:Wait()
end
end
end

function Breaker(part)
local label = part:WaitForChild("SurfaceGui"):WaitForChild("Frame"):WaitForChild("Code")
local function run()
task.wait(0.05)
if not Toggles.AutoBreaker.Value then return end
local target = tonumber(label.Text)
if target then
for _, v in part:GetChildren() do
if v.Name == "BreakerSwitch" and v:GetAttribute("ID") == target then
local trans = part:WaitForChild("SurfaceGui"):WaitForChild("Frame"):WaitForChild("Code"):WaitForChild("Frame").BackgroundTransparency
local pc = v:FindFirstChild("PrismaticConstraint")
local light = v:FindFirstChild("Light")
local sound = v:FindFirstChild("Sound")
if trans == 0 then
if v:GetAttribute("Enabled") then return end
v:SetAttribute("Enabled", true)
if pc then pc.TargetPosition = -0.2 end
if light then
light.Material = Enum.Material.Neon
local spark = light:FindFirstChild("Spark", true)
if spark then spark:Emit(1) end
end
if sound then sound:Play() end
elseif trans == 1 then
if not v:GetAttribute("Enabled") then return end
v:SetAttribute("Enabled", false)
if pc then pc.TargetPosition = 0.2 end
if light then light.Material = Enum.Material.Glass end
if sound then sound:Play() end
end
break
end
end
end
end
label:GetPropertyChangedSignal("Text"):Connect(run)
run()
end

function CheckAndNotifyEntity(entity)
if not entity then return end
if NotifiedEntities[entity] then return end
if not (Toggles.NotifySpawn and Toggles.NotifySpawn.Value) then return end
local key = NotifyOptionKey[entity.Name]
if not key then return end
local selected = Options.Notify and Options.Notify.Value
if type(selected) ~= "table" or not selected[key] then return end
NotifiedEntities[entity] = true
local con
con = entity.AncestryChanged:Connect(function(_, parent)
if not parent then
NotifiedEntities[entity] = nil
if con then con:Disconnect() con = nil end
end
end)
table.insert(Connections, con)
Notify(NotifyNames[entity.Name] .. " has Spawned", 5)
end

function NexusESPApplyColor(kind, color)
    local c = NexusESPLibrary.Settings
    for obj, data in pairs(NexusESPLibrary.Cache) do
        if data and data.NexusKind == kind then
            data.Color = color
            data.Text.TextColor3 = color
            data.Box.FillColor = color
            data.Line.BackgroundColor3 = color
            data.Arrow.ImageColor3 = color
        end
    end
end

    _NexusOldAddESP(self, cfg)
    local data = self.Cache[cfg.Object]
    if data then data.NexusKind = cfg.NexusKind end
end

local function NexusEntityActive(names)
    for _, name in ipairs(names) do
        if workspace:FindFirstChild(name, true) then
            return true
        end
    end
    return false
end

local function NexusMonsterAntiEnabled()
    for toggleName in pairs(NexusAntiEntityDefs) do
        local toggle = Toggles[toggleName]
        if toggle and toggle.Value then
            return true
        end
    end
    return false
end

local NexusMonsterAntiForcedGodmode = false

local function NexusUpdateMonsterAnti()
    if not LocalPlayer.Character then return end
    local shouldProtect = false
    for toggleName, names in pairs(NexusAntiEntityDefs) do
        local toggle = Toggles[toggleName]
        if toggle and toggle.Value and NexusEntityActive(names) then
            shouldProtect = true
            break
        end
    end

    if shouldProtect then
        if Toggles.Godmode and not Toggles.Godmode.Value then
            pcall(function() Toggles.Godmode:SetValue(true) end)
            NexusMonsterAntiForcedGodmode = true
        end
        -- Extra local touch protection for physical entities.
        for toggleName, names in pairs(NexusAntiEntityDefs) do
            local toggle = Toggles[toggleName]
            if toggle and toggle.Value then
                for _, name in ipairs(names) do
                    local obj = workspace:FindFirstChild(name, true)
                    if obj and obj:IsA("Model") then
                        for _, part in ipairs(obj:GetDescendants()) do
                            if part:IsA("BasePart") then
                                part.CanTouch = false
                            end
                        end
                    end
                end
            end
        end
    elseif NexusMonsterAntiForcedGodmode then
        pcall(function() Toggles.Godmode:SetValue(false) end)
        NexusMonsterAntiForcedGodmode = false
    end
end

-- Add only the missing toggles; original Anti Screech/A90/Halt/Dread/etc. stay intact.
for toggleName in pairs(NexusAntiEntityDefs) do
    if not Toggles[toggleName] then
        local label = toggleName:gsub("([a-z])([A-Z])", "%1 %2")
        label = "Anti " .. label:gsub("^Anti", "")
        BypassEntityBox:AddToggle(toggleName, {
            Text = label,
            Default = false,
            Callback = function()
                task.defer(NexusUpdateMonsterAnti)
            end,
        })
    end
end

function InfPrompt(Prompt)
local Char = LocalPlayer.Character
if not Char then return end
local RootPart = Char:FindFirstChild("HumanoidRootPart")
if not RootPart then return end
local Tool = Char:FindFirstChild("Lockpick") or Char:FindFirstChild("SkeletonKey") or Char:FindFirstChild("Shears")
local Name = Tool and Tool.Name
local TargetPart = Prompt.Parent and (Prompt.Parent:IsA("BasePart") and Prompt.Parent or Prompt.Parent:FindFirstChildWhichIsA("BasePart"))
local Dist = TargetPart and GetDistanceToPlayer(TargetPart.Position) or 999
if Tool and Dist <= (Prompt.MaxActivationDistance + 5) then
if Prompt:GetAttribute("InfItems") and Prompt:GetAttribute("Tool") ~= Name then
RestorePrompt(Prompt)
end
if Prompt:GetAttribute("InfItems") then
local startTime = Prompt:GetAttribute("InfTime") or 0
if (tick() - startTime > 8) or (Dist > Prompt.MaxActivationDistance + 7) then
RestorePrompt(Prompt)
end
return
end
if not Prompt:GetAttribute("InfItems") then
Prompt.Enabled = false
Prompt:SetAttribute("InfItems", true)
Prompt:SetAttribute("Tool", Name)
Prompt:SetAttribute("InfTime", tick())
Prompt.ClickablePrompt = false
local Clone = Prompt:Clone()
Clone.Name = "InfPrompt"
Clone.MaxActivationDistance = Prompt.MaxActivationDistance
Clone.Parent = Prompt.Parent
Clone.Enabled = true
Clone.ClickablePrompt = true

    local function GetToggle(name)
        return Toggles and Toggles[name]

    local function SoftColor(on)
        return on and Color3.fromRGB(70, 57, 101) or Color3.fromRGB(30, 31, 39)
    end

    local function BuildFeature(name, order)
        local Button = Instance.new("TextButton")
        Button.Name = name
        Button.LayoutOrder = order
        Button.AutoButtonColor = false
        Button.Text = ""
        Button.BackgroundColor3 = Color3.fromRGB(30, 31, 39)
        Button.BorderSizePixel = 0
        Button.Parent = FeatureScroll
        local Corner = Instance.new("UICorner")
        Corner.CornerRadius = UDim.new(0, 13)
        Corner.Parent = Button
        local Stroke = Instance.new("UIStroke")
        Stroke.Color = Color3.fromRGB(64, 65, 76)
        Stroke.Transparency = 0.58
        Stroke.Thickness = 1
        Stroke.Parent = Button

        local NameLabel = Instance.new("TextLabel")
        NameLabel.BackgroundTransparency = 1
        NameLabel.Position = UDim2.new(0, 14, 0, 9)
        NameLabel.Size = UDim2.new(1, -76, 0, 22)
        NameLabel.Font = Enum.Font.GothamMedium
        NameLabel.TextSize = 12
        NameLabel.TextColor3 = Color3.fromRGB(225, 224, 233)
        NameLabel.TextXAlignment = Enum.TextXAlignment.Left
        NameLabel.Text = Labels[name] or name
        NameLabel.Parent = Button

        local StateLabel = Instance.new("TextLabel")
        StateLabel.BackgroundTransparency = 1
        StateLabel.Position = UDim2.new(0, 14, 0, 34)
        StateLabel.Size = UDim2.new(1, -76, 0, 17)
        StateLabel.Font = Enum.Font.Gotham
        StateLabel.TextSize = 10
        StateLabel.TextColor3 = Color3.fromRGB(118, 120, 133)
        StateLabel.TextXAlignment = Enum.TextXAlignment.Left
        StateLabel.Text = "Unavailable"
        StateLabel.Parent = Button

        local Dot = Instance.new("Frame")
        Dot.Size = UDim2.new(0, 10, 0, 10)
        Dot.Position = UDim2.new(1, -28, 0.5, -5)
        Dot.BorderSizePixel = 0
        Dot.BackgroundColor3 = Color3.fromRGB(82, 84, 96)
        Dot.Parent = Button
        local DotCorner = Instance.new("UICorner")
        DotCorner.CornerRadius = UDim.new(1, 0)
        DotCorner.Parent = Dot

        local ToggleObject = GetToggle(name)
        local function Refresh()
            ToggleObject = GetToggle(name)
            local enabled = ToggleObject and ToggleObject.Value == true
            Button.BackgroundColor3 = SoftColor(enabled)
            Dot.BackgroundColor3 = enabled and Color3.fromRGB(196, 170, 255) or Color3.fromRGB(82, 84, 96)
            StateLabel.Text = ToggleObject and (enabled and "Enabled" or "Disabled") or "Unavailable"
            StateLabel.TextColor3 = enabled and Color3.fromRGB(177, 161, 210) or Color3.fromRGB(118, 120, 133)
        end
        Refresh()

        Button.MouseEnter:Connect(function()
            SoftTween:Create(Button, TweenInfo.new(0.14), {BackgroundTransparency = 0.06}):Play()
        end)
        Button.MouseLeave:Connect(function()
            SoftTween:Create(Button, TweenInfo.new(0.14), {BackgroundTransparency = 0}):Play()
        end)
        Button.MouseButton1Click:Connect(function()
            local T = GetToggle(name)
            if not T then
                task.wait()
                T = GetToggle(name)
            end

            if T then
                local current = (T.Value == true)
                pcall(function()
                    T:SetValue(not current)
                end)
                if name == "Door" or name == "Objective" or name == "HidingPlace" or name == "GateLever" or name == "Books" or name == "Breakers" or name == "Gold" or name == "Entity" or name == "Ladder" or name == "Fuse" then
                    task.defer(function() NexusSetESP(name, not current) end)
                end
                task.defer(function()
                    task.wait()
                    Refresh()
                end)
            else
                StateLabel.Text = "Unavailable"
            end
        end)
        if ToggleObject and ToggleObject.OnChanged then
            pcall(function() ToggleObject:OnChanged(Refresh) end)
        end
        return Button
    end

    local function ClearFeatures()
        for _, child in ipairs(FeatureScroll:GetChildren()) do
            if child:IsA("TextButton") then child:Destroy() end
        end
    end

    local function ShowPage(page)
        CurrentPage = page
        PageTitle.Text = string.upper(page)
        local count = #(Pages[page] or {})
        PageHint.Text = tostring(count) .. " features  •  soft controls"
        ClearFeatures()
        for i, name in ipairs(Pages[page] or {}) do
            BuildFeature(name, i)
        end
        for name, btn in pairs(NavButtons) do
            local active = name == page
            btn.BackgroundColor3 = active and Color3.fromRGB(53, 46, 70) or Color3.fromRGB(26, 27, 33)
            btn.TextColor3 = active and Color3.fromRGB(230, 221, 248) or Color3.fromRGB(151, 153, 165)
        end
    end

    local dragging = false
    local dragStart
    local startPos
    Top.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = Root.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    SoftInput.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            Root.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

        end)
    end
