local function IsPlayerCharacter(obj)
    if not obj then return false end
    for _,p in ipairs(Players:GetPlayers()) do
        if p.Character and obj:IsDescendantOf(p.Character) then return true end
    end
    return false
end

local function IsToolOrAccessory(obj)
    if not obj then return false end
    local current = obj
    while current and current ~= Workspace do
        if current:IsA("Tool") or current:IsA("Accessory") then return true end
        current = current.Parent
    end
    return false
end

local function CleanName(name)
    if not name then return "" end
    return string.lower(string.gsub(name, "[%s_%-%./]", ""))
end

local function GetMonsterType(obj)
    if not obj then return nil end
    if IsPlayerCharacter(obj) or IsToolOrAccessory(obj) then return nil end

    local function checkString(str)
        if not str then return nil end
        local cleanStr = CleanName(str)
        for name, monster in pairs(MonsterNames) do
            if cleanStr == CleanName(name) or string.find(cleanStr, CleanName(name), 1, true) then
                return monster
            end
        end
        return nil
    end

    local result = checkString(obj.Name)
    if result then return result end

    if obj:IsA("Model") then
        local lower = string.lower(obj.Name)
        if string.find(lower, "figure") or string.find(lower, "rush") or string.find(lower, "ambush") or string.find(lower, "seek") or string.find(lower, "screech") or string.find(lower, "eyes") or string.find(lower, "halt") then
            for name, monster in pairs(MonsterNames) do
                if string.find(lower, string.lower(name), 1, true) then
                    return monster
                end
            end
            return "MONSTER"
        end
    end

    return nil
end

local function IsBookObject(obj)
    if not obj then return false end
    local current = obj
    while current and current ~= Workspace do
        local n = string.lower(current.Name)
        if string.find(n, "book", 1, true) or string.find(n, "shelf", 1, true) then return true end
        current = current.Parent
    end
    return false
end

local function GetItemType(obj)
    if not obj or IsPlayerCharacter(obj) or IsToolOrAccessory(obj) or IsBookObject(obj) then return nil end
    local n = string.lower(obj.Name)

    for _, keyword in ipairs(ItemKeywords) do
        if string.find(n, keyword, 1, true) then
            if string.find(n, "gold", 1, true) or string.find(n, "coin", 1, true) or string.find(n, "dollar", 1, true) or string.find(n, "money", 1, true) then
                return "GOLD"
            elseif string.find(n, "key", 1, true) then
                return "KEY"
            elseif string.find(n, "crucifix", 1, true) then
                return "CRUCIFIX"
            elseif string.find(n, "vitamin", 1, true) then
                return "VITAMINS"
            else
                return "ITEM"
            end
        end
    end
    return nil
end

local function GetChestOrDrawerType(obj)
    if not obj or IsPlayerCharacter(obj) or IsToolOrAccessory(obj) then return nil end
    local n = string.lower(obj.Name)

    for _, keyword in ipairs(ChestKeywords) do
        if string.find(n, keyword, 1, true) then
            if string.find(n, "drawer", 1, true) then return "DRAWER"
            elseif string.find(n, "chest", 1, true) then return "CHEST"
            elseif string.find(n, "safe", 1, true) then return "SAFE"
            else return "CONTAINER"
            end
        end
    end
    return nil
end

local function GetBreakerType(obj)
    if not obj or IsPlayerCharacter(obj) or IsToolOrAccessory(obj) then return nil end
    local n = string.lower(obj.Name)

    for _, keyword in ipairs(BreakerKeywords) do
        if string.find(n, keyword, 1, true) then
            return "BREAKER"
        end
    end
    return nil
end

local function GetPart(obj)
    if not obj then return nil end
    if obj:IsA("BasePart") then return obj end
    if obj:IsA("Model") then
        if obj.PrimaryPart and obj.PrimaryPart:IsA("BasePart") then return obj.PrimaryPart end
        local hrp = obj:FindFirstChild("HumanoidRootPart", true)
        if hrp and hrp:IsA("BasePart") then return hrp end
        return obj:FindFirstChildWhichIsA("BasePart", true)
    end
    return nil
end

local function RemoveESP(obj)
    local data = ESPObjects[obj]
    if not data then return end
    if data.Box then data.Box:Destroy() end
    if data.Billboard then data.Billboard:Destroy() end
    ESPObjects[obj] = nil
end

local function CreateESP(obj, displayName, color)
    if not ESP_ENABLED or not obj or not obj.Parent or ESPObjects[obj] then return end
    local part = GetPart(obj)
    if not part then return end

    local Box = Instance.new("SelectionBox")
    Box.Name = "NexusBoxESP"
    Box.Adornee = obj
    Box.Color3 = color
    Box.LineThickness = 0.03
    Box.SurfaceTransparency = 0.85
    Box.Parent = Gui

    local Billboard = Instance.new("BillboardGui")
    Billboard.Name = "NexusESPLabel"
    Billboard.Adornee = part
    Billboard.Size = UDim2.new(0,120,0,24)
    Billboard.StudsOffset = Vector3.new(0,2,0)
    Billboard.AlwaysOnTop = true
    Billboard.Parent = Gui

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(1,0,1,0)
    Label.BackgroundTransparency = 1
    Label.Text = displayName
    Label.TextColor3 = color
    Label.TextStrokeTransparency = 0.1
    Label.Font = Enum.Font.GothamBold
    Label.TextSize = 10
    Label.Parent = Billboard

    ESPObjects[obj] = {
        Box = Box,
        Billboard = Billboard,
        Part = part,
    }
end

local function ClearESP()
    for obj in pairs(ESPObjects) do RemoveESP(obj) end
    table.clear(ESPObjects)
end

local function ScanObject(obj)
    if not ESP_ENABLED or not obj or not obj.Parent then return end

    local monster = GetMonsterType(obj)
    if monster then
        local target = obj
        if not obj:IsA("Model") then
            local current = obj.Parent
            local depth = 0
            while current and current ~= Workspace and depth < 6 do
                if current:IsA("Model") and GetMonsterType(current) then
                    target = current
                    break
                end
                current = current.Parent
                depth += 1
            end
        else
            target = obj
        end
        CreateESP(target, monster, MonsterColors[monster] or C.bad)
        return
    end

    local breaker = GetBreakerType(obj)
    if breaker then
        CreateESP(obj, "BREAKER", C.yellow)
        return
    end

    local item = GetItemType(obj)
    if item then
        local col = (item == "GOLD" or item == "KEY") and C.yellow or C.cyan
        CreateESP(obj, item, col)
        return
    end

    local container = GetChestOrDrawerType(obj)
    if container then
        CreateESP(obj, container, C.accent)
        return
    end
end

local function ScanWorkspace()
    if not ESP_ENABLED then return end
    for _, obj in ipairs(Workspace:GetDescendants()) do ScanObject(obj) end
end

Workspace.DescendantAdded:Connect(function(obj)
    if not ESP_ENABLED then return end
    task.defer(function() if obj.Parent then ScanObject(obj) end end)
end)

Workspace.DescendantRemoving:Connect(function(obj) RemoveESP(obj) end)

local EspTimer = 0
local function UpdateESP(dt)
    EspTimer += dt
    if EspTimer < 0.20 then return end
    EspTimer = 0

    if not ESP_ENABLED then return end
    for obj, data in pairs(ESPObjects) do
        if not obj or not obj.Parent then
            RemoveESP(obj)
        else
            local part = data.Part
            if not part or not part.Parent then
                RemoveESP(obj)
            elseif Root and Root.Parent then
                local dist = (part.Position - Root.Position).Magnitude
                local en = dist <= ESP_DISTANCE
                if data.Box then data.Box.Visible = en end
                if data.Billboard then data.Billboard.Enabled = en end
            end
        end
    end
end

local function FindNearestMonster()
    if not Root or not Root.Parent then return nil, nil end
    local closest, closestDistance = nil, nil
    for obj, data in pairs(ESPObjects) do
        if obj and obj.Parent and data.Part then
            local monster = GetMonsterType(obj)
            if monster then
                local d = (data.Part.Position - Root.Position).Magnitude
                if d <= 120 and (not closestDistance or d < closestDistance) then
                    closest = monster
                    closestDistance = d
                end
            end
        end
    end
    return closest, closestDistance
end

            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end
end)

RunService.RenderStepped:Connect(function(dt)
    if not Gui.Parent then return end
    Camera = Workspace.CurrentCamera

    -- Custom FOV Logic
    if FOV_ENABLED and Camera then
        Camera.FieldOfView = 95
    end

    -- Crosshair Logic
    CrosshairFrame.Visible = CROSSHAIR_ENABLED

    if not Character or not Character.Parent then return end
    Root = Character:FindFirstChild("HumanoidRootPart") or Root
    local Hum = GetHumanoid()

    if Hum and SpeedEnabled and math.abs(Hum.WalkSpeed - WALK_SPEED) > .5 then
        Hum.WalkSpeed = WALK_SPEED
    end

    if CrouchToggled and Hum then
        Hum.CameraOffset = Vector3.new(0, -1.5, 0)
    elseif Hum then
        Hum.CameraOffset = Vector3.zero
    end

    if FlyEnabled and Root and Root.Parent and Camera then
        if not FlyVelocity or not FlyVelocity.Parent then
            FlyVelocity = Instance.new("BodyVelocity")
            FlyVelocity.Name = "NexusFlyVelocity"
            FlyVelocity.MaxForce = Vector3.new(60000, 60000, 60000)
            FlyVelocity.P = 9000
            FlyVelocity.Velocity = Vector3.zero
            FlyVelocity.Parent = Root
        end

        local Move = Vector3.zero
        if UIS:IsKeyDown(Enum.KeyCode.W) then Move += Camera.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.S) then Move -= Camera.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.A) then Move -= Camera.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.D) then Move += Camera.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then Move += Vector3.yAxis end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then Move -= Vector3.yAxis end

        if Move.Magnitude > 0 then
            FlyVelocity.Velocity = Move.Unit * FLY_SPEED
        else
            FlyVelocity.Velocity = Vector3.zero
        end
    elseif FlyVelocity then
        FlyVelocity:Destroy()
        FlyVelocity = nil
    end

    UpdateESP(dt)

    if ENTITY_WARNING then
        local monster, distance = FindNearestMonster()
        if monster then
            Warning.Visible = true
            Warning.Text = "⚠ " .. tostring(monster) .. " • " .. string.format("%.0f", distance) .. " studs"
        else
            Warning.Visible = false
        end
    else
        Warning.Visible = false
    end
end)

Player.CharacterAdded:Connect(function(NewCharacter)
    Character = NewCharacter
    Root = NewCharacter:WaitForChild("HumanoidRootPart")
    Humanoid = NewCharacter:WaitForChild("Humanoid")
    CrouchToggled = false

    task.wait(.2)
    if SpeedEnabled then Humanoid.WalkSpeed = WALK_SPEED end
    if FlyVelocity then FlyVelocity:Destroy(); FlyVelocity = nil end

    if ESP_ENABLED then
        task.delay(.5, function()
            if ESP_ENABLED and Gui.Parent then
                ClearESP()
                ScanWorkspace()
            end
        end)
    end
end)

UIS.InputBegan:Connect(function(input, processed)
    if input.KeyCode == Enum.KeyCode.RightShift then
        Main.Visible = not Main.Visible
    elseif input.KeyCode == Enum.KeyCode.LeftControl and not processed then
        CrouchToggled = not CrouchToggled
    end
end)

Close.MouseButton1Click:Connect(function()
    ClearESP()
    if FlyVelocity then FlyVelocity:Destroy() end
    if SpinConnection then SpinConnection:Disconnect() end
    Gui:Destroy()
end)
